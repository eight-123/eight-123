const CHROMATIC = [ "C","Des","D","Es","E", "F","Ges","G","As","A","H","B" ]; const SOLFEGE = [ "ド","レ","ミ","ファ","ソ","ラ","シ" ]; const DIATONIC_SCALES = { inC: ["C","D","E","F","G","A","H"], inB: ["B","C","D","Es","F","G","A"], inEb: ["Es","F","G","As","B","C","D"], inF: ["F","G","A","B","C","D","E"] }; function convert(){ // ===== 入力 ===== const key = document.getElementById("instrument").value; const solfege = document.getElementById("note").value; const accidental = Number( document.getElementById("accidental").value ); // ===== 調の全音階取得 ===== const scale = DIATONIC_SCALES[key]; // ===== A：ドレミの番号 ===== const A = SOLFEGE.indexOf(solfege); // ===== B：その調の音 ===== const B = scale[A]; // ===== C：半音階位置 ===== let C = CHROMATIC.indexOf(B); // ===== D：臨時記号 ===== let D = C + accidental; D = (D + CHROMATIC.length) % CHROMATIC.length; // ===== 結果 ===== const result = CHROMATIC[D]; document.getElementById("result") .textContent = result; }
